/**
 * GET /api/analytics/players/:playerId
 *
 * Returns player's champion pool with stats
 *
 * Response:
 * {
 *   playerId: string,
 *   playerName: string,
 *   championPool: Array<{
 *     champion: string,
 *     gamesPlayed: number,
 *     winRate: number,
 *     comfortLevel: 'signature' | 'comfort' | 'recent' | 'historical'
 *     roles: string[]
 *   }>,
 *   meta: { responseTime: number }
 * }
 */

import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'

export const runtime = 'edge'

export async function GET(
  request: NextRequest,
  { params }: { params: { playerId: string } }
) {
  const startTime = Date.now()
  const { playerId } = params

  try {
    const supabase = await createClient()

    // Query player_champion_pools_computed from Phase 2
    const { data, error } = await supabase
      .from('synapse.player_champion_pools_computed')
      .select('*')
      .eq('player_id', playerId)
      .order('games_played', { ascending: false })
      .limit(20)

    if (error) {
      console.error('[GET /api/analytics/players] Error:', error)
      return NextResponse.json({ error: 'Failed to fetch player pool' }, { status: 500 })
    }

    // Transform to API response format
    const championPool = (data || []).map(row => ({
      champion: row.champion_id, // Will need champion name lookup
      gamesPlayed: row.games_played,
      winRate: row.smoothed_win_rate,
      comfortLevel: classifyComfortLevel(row.games_played, row.smoothed_win_rate),
      roles: row.roles_played || []
    }))

    return NextResponse.json({
      playerId,
      playerName: data?.[0]?.player_name || 'Unknown',
      championPool,
      meta: { responseTime: Date.now() - startTime }
    })

  } catch (error) {
    console.error('[GET /api/analytics/players] Error:', error)
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
  }
}

function classifyComfortLevel(
  gamesPlayed: number,
  winRate: number
): 'signature' | 'comfort' | 'recent' | 'historical' {
  if (gamesPlayed >= 10 && winRate >= 0.55) return 'signature'
  if (gamesPlayed >= 5 && winRate >= 0.50) return 'comfort'
  if (gamesPlayed >= 3) return 'recent'
  return 'historical'
}

'use client'

import { useState, useEffect } from 'react'

interface ChampionPoolEntry {
  champion: string
  gamesPlayed: number
  winRate: number
  comfortLevel: 'signature' | 'comfort' | 'recent' | 'historical'
  roles: string[]
}

interface PlayerPoolData {
  playerId: string
  playerName: string
  championPool: ChampionPoolEntry[]
}

export function usePlayerPool(playerId: string | null) {
  const [data, setData] = useState<PlayerPoolData | null>(null)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<Error | null>(null)

  useEffect(() => {
    if (!playerId) {
      setData(null)
      return
    }

    const fetchPool = async () => {
      setLoading(true)
      setError(null)

      try {
        const res = await fetch(`/api/analytics/players/${playerId}`)
        if (!res.ok) throw new Error(`HTTP ${res.status}`)
        const json = await res.json()
        setData(json)
      } catch (e) {
        setError(e as Error)
      } finally {
        setLoading(false)
      }
    }

    fetchPool()
  }, [playerId])

  return { data, loading, error }
}

'use client'

import { usePlayerPool } from '@/lib/hooks/use-player-pool'
import { useDraftStore } from '@/lib/draft/store'

interface PlayerPoolPanelProps {
  playerId: string | null
  playerName: string | null
  role: string
}

export function PlayerPoolPanel({ playerId, playerName, role }: PlayerPoolPanelProps) {
  const { data, loading, error } = usePlayerPool(playerId)
  const availableChampions = useDraftStore((state) => state.availableChampions)

  if (!playerId) {
    return (
      <div className="bg-gray-800 rounded-lg p-4 text-center text-gray-400">
        <p>Select a player to view their champion pool</p>
      </div>
    )
  }

  if (loading) {
    return (
      <div className="bg-gray-800 rounded-lg p-4 text-center">
        <div className="animate-pulse">
          <div className="h-4 bg-gray-700 rounded w-3/4 mx-auto mb-2"></div>
          <div className="h-4 bg-gray-700 rounded w-1/2 mx-auto"></div>
        </div>
      </div>
    )
  }

  if (error || !data) {
    return (
      <div className="bg-gray-800 rounded-lg p-4 text-center text-red-400">
        <p>Failed to load player data</p>
      </div>
    )
  }

  // Group champions by comfort level
  const signature = data.championPool.filter(c => c.comfortLevel === 'signature')
  const comfort = data.championPool.filter(c => c.comfortLevel === 'comfort')
  const others = data.championPool.filter(c =>
    c.comfortLevel !== 'signature' && c.comfortLevel !== 'comfort'
  )

  return (
    <div className="bg-gray-800 rounded-lg p-4">
      <h3 className="text-lg font-semibold text-white mb-1">
        {playerName || data.playerName}
      </h3>
      <p className="text-sm text-gray-400 mb-4 uppercase">{role}</p>

      {/* Signature Champions */}
      {signature.length > 0 && (
        <div className="mb-4">
          <h4 className="text-sm font-semibold text-amber-400 mb-2 flex items-center gap-2">
            <span>⭐</span> Signature Champions
          </h4>
          <div className="space-y-2">
            {signature.map((champ) => (
              <ChampionPoolEntry
                key={champ.champion}
                {...champ}
                isAvailable={availableChampions.has(champ.champion)}
              />
            ))}
          </div>
        </div>
      )}

      {/* Comfort Champions */}
      {comfort.length > 0 && (
        <div className="mb-4">
          <h4 className="text-sm font-semibold text-green-400 mb-2 flex items-center gap-2">
            <span>✓</span> Comfort Picks
          </h4>
          <div className="space-y-2">
            {comfort.map((champ) => (
              <ChampionPoolEntry
                key={champ.champion}
                {...champ}
                isAvailable={availableChampions.has(champ.champion)}
              />
            ))}
          </div>
        </div>
      )}

      {/* Other Champions */}
      {others.length > 0 && (
        <div>
          <h4 className="text-sm font-semibold text-gray-400 mb-2">
            Other Picks
          </h4>
          <div className="space-y-2">
            {others.slice(0, 5).map((champ) => (
              <ChampionPoolEntry
                key={champ.champion}
                {...champ}
                isAvailable={availableChampions.has(champ.champion)}
              />
            ))}
          </div>
        </div>
      )}

      {data.championPool.length === 0 && (
        <p className="text-gray-400 text-sm">No champion pool data available</p>
      )}
    </div>
  )
}

// Individual champion entry
function ChampionPoolEntry({
  champion,
  gamesPlayed,
  winRate,
  comfortLevel,
  roles,
  isAvailable,
}: {
  champion: string
  gamesPlayed: number
  winRate: number
  comfortLevel: string
  roles: string[]
  isAvailable: boolean
}) {
  return (
    <div className={`
      flex items-center justify-between p-2 rounded
      ${isAvailable ? 'bg-gray-700' : 'bg-gray-700/50 opacity-50'}
    `}>
      <div className="flex items-center gap-2">
        <div className="w-8 h-8 bg-gray-600 rounded flex items-center justify-center text-sm font-bold">
          {champion.charAt(0)}
        </div>
        <div>
          <div className="text-white text-sm font-medium">{champion}</div>
          {roles.length > 1 && (
            <div className="text-xs text-amber-400">Flex: {roles.join('/')}</div>
          )}
        </div>
      </div>

      <div className="text-right">
        <div className={`text-sm font-semibold ${winRate >= 0.55 ? 'text-green-400' : winRate >= 0.50 ? 'text-white' : 'text-red-400'}`}>
          {(winRate * 100).toFixed(0)}%
        </div>
        <div className="text-xs text-gray-400">{gamesPlayed} games</div>
      </div>

      {!isAvailable && (
        <div className="absolute right-2 text-xs text-red-400">BANNED</div>
      )}
    </div>
  )
}

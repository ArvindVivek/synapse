**champion-aliases.ts:**
Champion name normalization utility:

```typescript
export const CHAMPION_ALIASES: Record<string, string[]> = {
  'Wukong': ['MonkeyKing', 'Kong'],
  'TwistedFate': ['TF', 'Twisted Fate'],
  'MissFortune': ['MF', 'Miss Fortune'],
  'JarvanIV': ['J4', 'Jarvan IV', 'Jarvan'],
  'RekSai': ["Rek'Sai", 'Reksai'],
  'KhaZix': ["Kha'Zix", 'Khazix'],
  'VelKoz': ["Vel'Koz", 'Velkoz'],
  'ChoGath': ["Cho'Gath", 'Chogath'],
  'KogMaw': ["Kog'Maw", 'Kogmaw'],
  // etc. - cover all champions with apostrophes or multi-word names
};

export function normalizeChampionName(name: string): string {
  // Check if name matches any alias
  for (const [canonical, aliases] of Object.entries(CHAMPION_ALIASES)) {
    if (aliases.some(a => a.toLowerCase() === name.toLowerCase())) {
      return canonical;
    }
    if (canonical.toLowerCase() === name.toLowerCase()) {
      return canonical;
    }
  }
  // Return original (removing spaces, apostrophes for consistency)
  return name.replace(/['\s]/g, '');
}
```

```typescript
// app/api/draft/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { v4 as uuidv4 } from 'uuid'

interface CreateDraftRequest {
  userSide: 'blue' | 'red'
  blueTeamId?: string
  redTeamId?: string
  // Optional: pre-populate with specific champions
  champions?: string[]
}

export async function POST(request: NextRequest) {
  const startTime = Date.now()

  try {
    const body: CreateDraftRequest = await request.json()

    // Validate request
    if (!body.userSide || !['blue', 'red'].includes(body.userSide)) {
      return NextResponse.json(
        { error: 'Invalid userSide, must be "blue" or "red"' },
        { status: 400 }
      )
    }

    // Generate draft ID
    const draftId = uuidv4()

    // Get champion list from database (or use provided list)
    const supabase = createClient()
    let champions: string[] = body.champions || []

    if (champions.length === 0) {
      const { data: championData } = await supabase
        .from('champions')
        .select('name')
        .order('name')

      champions = championData?.map(c => c.name) || []
    }

    // Return draft session info
    const response = {
      id: draftId,
      userSide: body.userSide,
      blueTeamId: body.blueTeamId || null,
      redTeamId: body.redTeamId || null,
      champions,
      createdAt: new Date().toISOString()
    }

    const duration = Date.now() - startTime
    console.log(`[API] POST /api/draft completed in ${duration}ms`)

    return NextResponse.json(response, {
      status: 201,
      headers: { 'X-Response-Time': `${duration}ms` }
    })
  } catch (error) {
    console.error('[API] POST /api/draft error:', error)
    return NextResponse.json(
      { error: 'Failed to create draft session' },
      { status: 500 }
    )
  }
}
```

```typescript
// In-memory draft state storage (for hackathon - would use Redis/DB in production)
// This is a simple Map that stores draft states by ID
export const draftStates = new Map<string, DraftState>()

export async function GET(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  const startTime = Date.now()
  const { id } = params

  const state = draftStates.get(id)

  if (!state) {
    return NextResponse.json(
      { error: 'Draft not found' },
      { status: 404 }
    )
  }

  // Convert Set to array for JSON serialization
  const response = {
    ...state,
    availableChampions: Array.from(state.availableChampions)
  }

  const duration = Date.now() - startTime
  return NextResponse.json(response, {
    headers: { 'X-Response-Time': `${duration}ms` }
  })
}
```

**POST /api/draft/:id/action (Execute action):**
```typescript
// app/api/draft/[id]/action/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { validateAction, ValidationErrorCode } from '@/lib/draft/validation'
import { getTurnInfo } from '@/lib/draft/sequence'
import { DraftAction, DraftState } from '@/lib/draft/types'
import { draftStates } from '../route'

interface ActionRequest {
  type: 'BAN' | 'PICK' | 'UNDO' | 'RESET'
  champion?: string
  role?: string
}

export async function POST(
  request: NextRequest,
  { params }: { params: { id: string } }
) {
  const startTime = Date.now()
  const { id } = params

  try {
    const body: ActionRequest = await request.json()
    const state = draftStates.get(id)

    if (!state) {
      return NextResponse.json(
        { error: 'Draft not found' },
        { status: 404 }
      )
    }

    // Build action
    const action: DraftAction = body.type === 'BAN'
      ? { type: 'BAN', champion: body.champion! }
      : body.type === 'PICK'
      ? { type: 'PICK', champion: body.champion!, role: body.role as any }
      : { type: body.type }

    // Validate action
    const validation = validateAction(state, action)

    if (!validation.valid) {
      return NextResponse.json(
        {
          error: validation.error.message,
          code: validation.error.code,
          details: validation.error.details
        },
        { status: 400 }
      )
    }

    // Execute action (mutate state)
    const newState = executeAction(state, action)
    draftStates.set(id, newState)

    const duration = Date.now() - startTime
    return NextResponse.json(
      {
        success: true,
        state: {
          ...newState,
          availableChampions: Array.from(newState.availableChampions)
        }
      },
      { headers: { 'X-Response-Time': `${duration}ms` } }
    )
  } catch (error) {
    console.error('[API] POST /api/draft/:id/action error:', error)
    return NextResponse.json(
      { error: 'Failed to execute action' },
      { status: 500 }
    )
  }
}

// Helper to execute action and return new state
function executeAction(state: DraftState, action: DraftAction): DraftState {
  // Clone state for immutability
  const newState: DraftState = {
    ...state,
    blue: { ...state.blue, bans: [...state.blue.bans], picks: [...state.blue.picks] },
    red: { ...state.red, bans: [...state.red.bans], picks: [...state.red.picks] },
    availableChampions: new Set(state.availableChampions)
  }

  if (action.type === 'BAN') {
    const turnInfo = getTurnInfo(newState.currentTurn)!
    if (turnInfo.side === 'blue') {
      newState.blue.bans.push(action.champion)
    } else {
      newState.red.bans.push(action.champion)
    }
    newState.availableChampions.delete(action.champion)
    newState.currentTurn++
    // Update phase if needed
    newState.phase = getPhaseForTurn(newState.currentTurn)
  }

  if (action.type === 'PICK') {
    const turnInfo = getTurnInfo(newState.currentTurn)!
    const pick = { champion: action.champion, role: action.role || null }
    if (turnInfo.side === 'blue') {
      newState.blue.picks.push(pick)
    } else {
      newState.red.picks.push(pick)
    }
    newState.availableChampions.delete(action.champion)
    newState.currentTurn++
    newState.phase = getPhaseForTurn(newState.currentTurn)

    // Check if draft complete
    if (newState.currentTurn > 20) {
      newState.isComplete = true
      newState.completedAt = new Date()
    }
  }

  // UNDO and RESET would need history tracking (simplified for MVP)

  return newState
}
```

Create Supabase Broadcast channel management for real-time draft sync:

**Broadcast vs Postgres Changes:**
- Broadcast: 50-100ms latency, ephemeral (no DB write)
- Postgres Changes: 150-300ms latency, persisted
- For draft: Use Broadcast (speed > persistence during draft)

**Implementation:**
```typescript
// lib/draft/realtime.ts
import { createBrowserClient } from '@supabase/ssr'
import { DraftAction, DraftState } from './types'

type DraftChannel = ReturnType<typeof createBrowserClient>['channel']

// Event types for broadcast
export type DraftEvent =
  | { type: 'ACTION'; action: DraftAction; state: DraftState; timestamp: number }
  | { type: 'SYNC_REQUEST'; requesterId: string }
  | { type: 'SYNC_RESPONSE'; state: DraftState; responderId: string }

// Create a broadcast channel for a draft session
export function createDraftChannel(
  supabase: ReturnType<typeof createBrowserClient>,
  draftId: string
): DraftChannel {
  return supabase.channel(`draft:${draftId}`, {
    config: {
      broadcast: { self: true }  // Receive own broadcasts for confirmation
    }
  })
}

// Broadcast an action to all connected clients
export async function broadcastAction(
  channel: DraftChannel,
  action: DraftAction,
  newState: DraftState
): Promise<void> {
  const event: DraftEvent = {
    type: 'ACTION',
    action,
    state: {
      ...newState,
      // Convert Set to array for serialization
      availableChampions: Array.from(newState.availableChampions) as any
    },
    timestamp: Date.now()
  }

  await channel.send({
    type: 'broadcast',
    event: 'draft_action',
    payload: event
  })
}

// Subscribe to draft updates
export function subscribeToDraft(
  channel: DraftChannel,
  onAction: (event: DraftEvent) => void,
  onSync?: (state: DraftState) => void
): () => void {
  channel
    .on('broadcast', { event: 'draft_action' }, ({ payload }) => {
      onAction(payload as DraftEvent)
    })
    .on('broadcast', { event: 'sync_response' }, ({ payload }) => {
      if (onSync) {
        const event = payload as DraftEvent
        if (event.type === 'SYNC_RESPONSE') {
          // Convert array back to Set
          const state = {
            ...event.state,
            availableChampions: new Set(event.state.availableChampions as unknown as string[])
          }
          onSync(state)
        }
      }
    })
    .subscribe()

  // Return unsubscribe function
  return () => {
    channel.unsubscribe()
  }
}

// Request sync from other clients (for late joiners)
export async function requestSync(
  channel: DraftChannel,
  requesterId: string
): Promise<void> {
  await channel.send({
    type: 'broadcast',
    event: 'sync_request',
    payload: { type: 'SYNC_REQUEST', requesterId }
  })
}

// Respond to sync request (host client only)
export async function respondToSync(
  channel: DraftChannel,
  state: DraftState,
  responderId: string
): Promise<void> {
  await channel.send({
    type: 'broadcast',
    event: 'sync_response',
    payload: {
      type: 'SYNC_RESPONSE',
      state: {
        ...state,
        availableChampions: Array.from(state.availableChampions)
      },
      responderId
    }
  })
}

// Presence tracking for draft participants
export interface DraftPresence {
  odentId: string
  side: 'blue' | 'red' | 'spectator'
  joinedAt: number
}

export function trackPresence(
  channel: DraftChannel,
  clientId: string,
  side: 'blue' | 'red' | 'spectator'
): void {
  channel.track({
    clientId,
    side,
    joinedAt: Date.now()
  })
}
```

**Key considerations:**

2. State serialization handles Set -> Array -> Set conversion
3. Sync mechanism for late joiners (request/response pattern)

```typescript
handleRemoteAction: (event: DraftEvent) => {
  if (event.type !== 'ACTION') return

  set((draft) => {
    // Apply remote state (trust the broadcaster)
    // Only apply if timestamp is newer than our last action
    const remoteState = event.state

    // Convert array back to Set for availableChampions
    draft.currentTurn = remoteState.currentTurn
    draft.phase = remoteState.phase
    draft.blue = remoteState.blue
    draft.red = remoteState.red
    draft.availableChampions = new Set(remoteState.availableChampions as unknown as string[])
    draft.isComplete = remoteState.isComplete
    draft.completedAt = remoteState.completedAt
    draft.lastValidationError = null
  })
}
```

```typescript
connectRealtime: (supabase, draftId) => {
  const channel = createDraftChannel(supabase, draftId)

  const unsubscribe = subscribeToDraft(
    channel,
    (event) => get().handleRemoteAction(event),
    (state) => {
      // Sync response - update full state
      set((draft) => {
        Object.assign(draft, state)
        draft.availableChampions = new Set(state.availableChampions as unknown as string[])
      })
    }
  )

  set((draft) => {
    draft.channel = channel
    draft.isConnected = true
  })

  // Return cleanup function
  return () => {
    unsubscribe()
    set((draft) => {
      draft.channel = null
      draft.isConnected = false
    })
  }
}
```

export interface WinRateBreakdown {
  baseComposition: number    // Champion base win rates contribution
  synergies: number          // Team synergy bonus/penalty
  matchups: number           // Head-to-head matchup advantage
  sideAdvantage: number      // Blue/red side modifier
}

export interface WinRateProjection {
  blueWinRate: number        // 0.05 to 0.95
  redWinRate: number         // 0.05 to 0.95 (always 1 - blueWinRate)
  breakdown: WinRateBreakdown
  confidence: 'high' | 'medium' | 'low'
  turnNumber: number         // Last updated turn
}

interface PickRecord {
  champion: string
  role: string | null
  side: 'blue' | 'red'
}

export class WinRateProjector {
  private projection: WinRateProjection
  private bluePicks: PickRecord[]
  private redPicks: PickRecord[]
  private patchVersion: string

    this.patchVersion = patchVersion
    this.bluePicks = []
    this.redPicks = []
    this.projection = {
      blueWinRate: 0.50,
      redWinRate: 0.50,
      breakdown: {
        baseComposition: 0,
        synergies: 0,
        matchups: 0,
        sideAdvantage: 0.02  // Blue side starts with +2% advantage
      },
      confidence: 'low',
      turnNumber: 0
    }
  }

  getCurrentProjection(): WinRateProjection {
    return { ...this.projection }
  }

  reset(): void {
    // Reset to initial state
  }
}

Implementation:
1. Add pick to internal tracking (bluePicks or redPicks)
2. Calculate delta for each breakdown component:

**Base Composition Delta:**
- Get champion's base win rate from champion_stats_computed
- Delta = (championWinRate - 0.50) * 0.20 (20% weight)
- Apply sign based on side (positive for blue, negative for red)

**Synergy Delta:**
- For each existing teammate, get synergy score
- Filter to synergies with games_together >= 5
- Sum: (synergyWinRate - 0.50) * 0.10 per pair (10% weight per synergy)
- Apply sign based on side

**Matchup Delta:**
- For each enemy pick, get matchup score
- Filter to matchups with games >= 3
- Sum: matchup_delta * 0.15 (15% weight per matchup)
- Apply sign based on side

3. Apply deltas to breakdown components (incremental, not replacement)
4. Clamp each component delta to max ±0.05 per pick (prevent volatility)
5. Recalculate final win rate: 0.50 + sum(breakdown components)
6. Clamp to [0.05, 0.95]
7. Update confidence based on turnNumber

private clampWinRate(value: number): number {
  // Never show 0% or 100%
  return Math.max(0.05, Math.min(0.95, value))
}

private clampDelta(delta: number, maxDelta: number = 0.05): number {
  // Limit per-pick impact to prevent wild swings
  return Math.max(-maxDelta, Math.min(maxDelta, delta))
}

private recalculateTotal(): void {
  // Sum all breakdown components + base 50%
  const { breakdown } = this.projection
  const blueTotal = 0.50 +
    breakdown.baseComposition +
    breakdown.synergies +
    breakdown.matchups +
    breakdown.sideAdvantage

  this.projection.blueWinRate = this.clampWinRate(blueTotal)
  this.projection.redWinRate = 1 - this.projection.blueWinRate
}

// Factory function for clean API
export function createWinRateProjector(patchVersion: string): WinRateProjector {
  return new WinRateProjector(patchVersion)
}

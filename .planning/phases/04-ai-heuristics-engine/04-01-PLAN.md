```typescript
// Scoring weights that sum to 1.0
interface ScoringWeights {
  synergy: number      // Team synergy contribution
  counter: number      // Matchup advantage
  composition: number  // Team comp balance (damage types, roles)
  side: number         // Blue/red side advantage
  flex: number         // Multi-role potential
}

// Score breakdown for transparency
interface ScoreBreakdown {
  synergy: number      // 0.0-1.0 normalized
  counter: number
  composition: number
  side: number
  flex: number
}

// Full recommendation with reasoning
interface PickRecommendation {
  champion: string
  totalScore: number
  scores: ScoreBreakdown
  confidence: 'high' | 'medium' | 'low'
  reasoning: string[]  // Human-readable explanations
}

// Draft context for scoring
interface DraftContext {
  currentTurn: number
  phase: 'ban1' | 'pick1' | 'ban2' | 'pick2'
  userSide: 'blue' | 'red'
  userPicks: string[]
  userBans: string[]
  opponentPicks: string[]
  opponentBans: string[]
  availableChampions: Set<string>
}

// Turn classification for weight selection
type TurnPhase = 'early' | 'mid' | 'late'
```

```typescript
import { getTeamSynergies } from '@/lib/queries/synergies'
import { calculateSynergyScore } from './scoring/synergy-score'
import { calculateCounterScore } from './scoring/counter-score'
import { calculateCompositionScore } from './scoring/composition-score'
import {
  ScoringWeights, PickRecommendation, DraftContext,
  EARLY_PICK_WEIGHTS, MID_PICK_WEIGHTS, LATE_PICK_WEIGHTS,
  getTurnPhase
} from './types'

export function getWeightsForTurn(turn: number): ScoringWeights {
  const phase = getTurnPhase(turn)
  switch (phase) {
    case 'early': return EARLY_PICK_WEIGHTS
    case 'mid': return MID_PICK_WEIGHTS
    case 'late': return LATE_PICK_WEIGHTS
  }
}

export async function scoreChampionForPick(
  champion: string,
  context: DraftContext,
  patchVersion: string
): Promise<PickRecommendation>
```

Implementation requirements:
1. Get weights using getWeightsForTurn(context.currentTurn)
2. Calculate each score component in parallel using Promise.all
3. For side score: Use simple heuristic (+0.02 for blue side if champion has >52% blue WR)
4. For flex score: Check if champion has role_confidence < 0.7 in champion stats (flex potential)
5. Apply weighted sum: totalScore = sum(score * weight) for all components
6. Set confidence based on data quality:
   - 'high' if all components have 10+ games of data
   - 'medium' if some components have 5-10 games
   - 'low' if sparse data
